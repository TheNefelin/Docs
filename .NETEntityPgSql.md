```csharp
public class AppDbContext : DbContext
{
    public AppDbContext(DbContextOptions<AppDbContext> options) : base(options) {}

    public DbSet<Usuario> Usuarios { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        // Configuraciones del modelo
        modelBuilder.Entity<Usuario>(entity =>
        {
            entity.HasKey(e => e.Id);
            entity.Property(e => e.Nombre).HasMaxLength(100);
            entity.HasIndex(e => e.Email).IsUnique();
        });
    }
}
```

## Controlador
```csharp
[Route("api/donations")]
[ApiController]
public class DonationsController : ControllerBase
{
    private readonly ILogger<DonationsController> _logger;
    private readonly AppDbContext _context;

    public DonationsController(ILogger<DonationsController> logger, AppDbContext context)
    {
        _logger = logger;
        _context = context;
    }

    [HttpGet("companies")]
    public async Task<ActionResult<IEnumerable<Company>>> Get()
    {
        var companies = await _context.companies.ToListAsync();
        return Ok(companies);
    }
}
```

```csharp
{
    private readonly IConfiguration _config;
    
    public MiServicio(IConfiguration config)
    {
        _config = config;
    }
    
    public string GetRegion()
    {
        // Esto lee de la MISMA configuración que builder.Configuration
        return _config["AWS_REGION"] ?? "us-east-1"; // valor por defecto
    }
}
```
